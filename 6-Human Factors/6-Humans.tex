\section{Perception}

\begin{frame}
	\frametitle{Perception}
	\begin{itemize}
		\item \textbf{First stage of processing} in the brain, occurs when sensory signals are received as input.  It is at this stage human makes meanings
		\item Perception has been studied in a area of experimental psychology known as \textbf{psychophysics} - examines the relationship between perception and physical phenomena
%		\item In psychophysics experiment, human is presented with physical stimulus and is then asked how they felt/perceive
%		\item A common experimental goal is to measure \textit{just noticeable difference} (JND) - does the two stimuli differ?  By manipulating the small difference between two stimuli, we can better understand human perception 
	\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Facts about Perception}
\begin{enumerate}
	\item Our perception is biased by
	\begin{itemize}
		\item our goals
		\item our belief
		\item our experience
		\item the context
	\end{itemize}
	\item Our vision is optimized to see structure
	\item Our color vision is limited
	\item Our peripheral vision is poor
	\item Visual search is linear unless target "pops"
	\item Reading is unnatural
\end{enumerate}
\end{frame}

\subsection{Design implications}

\begin{frame}
\frametitle{Design implications}
	\begin{itemize}
		\item\textbf{Don't believe what users say}; instead understand their goals and knowledge (and what they don't know)
		\item Always exploit \textbf{structure} rules
		\item Use \textbf{color} very carefully
		\item Humans can focus only at \textbf{very tiny spot}.  Thus put where users are \textbf{looking}
		\item Use \textbf{affordance}, \textbf{convention}, \textbf{constraints}, \textbf{mapping} can increase visual search speed
	\end{itemize}
\end{frame}

	\frametitle{Short-term memory}
	\begin{itemize}
		\item Also known as working memory
		\item Can remember around 7 ($\pm$2) unrelated items (3-5 are better estimates, according to Jeff)

\begin{frame}
	\frametitle{Long-term memory}
	\begin{itemize}
		\item Stores life time experience but prone to error and biases
		\item Similar experience trigger same patterns $>$ \textbf{recognition}
		\item Internal neural activity triggers pattern $>$ \textbf{recall}
		\item Why we forget?
		\begin{itemize}
			\item \textbf{Decay theory}: proposes that memory fades due to mere passage of time - active rehearsing information is believed to counter this temporal decline
			\item \textbf{Interference theory}: proposes that similar information can make memories less accessible
		\end{itemize}
	\end{itemize}

\subsection{Design implications}

\begin{frame}
	\frametitle{Design Implications}
		\begin{itemize}
			\item Use \textbf{convention} and \textbf{consistency}
			\item Use \textbf{recognition} instead of recall when possible (but do not forget ways for novice to transition to experts)
			\item Use \textbf{chunking} when possible; 3-5 rule
			\item \textbf{Put the knowledge in of the world} (Norman, 1988), e.g.,  bookmarks, history of commands, tagging, time stamping, reminders, marked emails, pwd etc.
		\end{itemize}
\end{frame}

\section{Personality}

\begin{frame}
\frametitle{Personality: Same app, different preferences}
\small
Two people use the same fitness app. They want different experiences.
\medskip
\begin{columns}[t]
\begin{column}{0.48\textwidth}
\begin{block}{``Let me compete.''}
\begin{itemize}
\item Share a run with friends.
\item Join a group challenge.
\item Compare progress with others.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{``Let me focus privately.''}
\begin{itemize}
\item Keep activity private.
\item Set a personal goal.
\item Compare with yesterday's self.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\medskip
\textbf{Ask:} Which would you prefer? Would your answer change with the situation?
\medskip

Personality describes broad tendencies. Goals, experience, and context also matter.
\end{frame}

\begin{frame}
\frametitle{Surprise: A reward can reduce interest}
\small
\textbf{Predict:} If someone enjoys drawing, will paying them for every picture always make them want to draw more?
\medskip
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{@{}p{4.8cm}p{4.8cm}@{}}
\toprule
\textbf{Before} & \textbf{After rewards become expected}\\
\midrule
``I draw because I enjoy it.'' & ``What do I get for drawing?''\\
\bottomrule
\end{tabular}
\medskip

A review of 128 experiments found that expected tangible rewards could reduce later voluntary engagement in already interesting activities.
\medskip

This does \textbf{not} mean all rewards are harmful. Reward type, feedback, and context matter.
\begin{block}{Remember}
More rewards do not automatically mean more motivation. Build interest and competence, not only points.
\end{block}
{\tiny Deci, Koestner, and Ryan (1999). Dialogue illustrates the mechanism, not a study quotation.}
\end{frame}

\subsection{Reasoning}

\begin{frame}
	\frametitle{System 1 and System 2}
	\begin{itemize}
		\item Our brain has two systems: system 1 and system 2
		\item \textbf{System 1} is the \textit{irrational} brain - fast, automatic, unconscious, yet govern most of our behavior
		\item \textbf{System 2} is the \textit{rational} brain - slow, precise, conscious, "believes" it governs our behaviors
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{System 1 and 2}
	\centering
	\includegraphics[width=0.8\linewidth]{image/brain}
\end{frame}

\begin{frame}
	\frametitle{System 1 and System 2}
	\begin{itemize}
		\item A baseball and a bat together cost 110.  The bat costs 100 more than the ball.
		How much does the ball cost? 
		\item System 1 instant answer: 10  (wrong)
		\item System 2 may reject that answer.  Or not.
		\item System 2 can calculate correct answer; System 1 cannot.
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Human decisions are rarely rational}
	System 1 usually controls decisions but is very biased.
	\begin{itemize}
		\item \textbf{Losses} mean more than \textbf{gains}
		\item \textbf{Recent} history and strong memories "feel" more
		\item \textbf{Experience} and \textbf{intuition} means more than mountains of statistics and data
		\item People \textbf{avoid risks} for potential gains, but \textbf{take risks} for potential losses
		\item Influence by \textbf{word} (75\% survival rate vs. 25\% mortality rate)
	\end{itemize}
\end{frame}

        \node[text=blue!70!black,font=\Large\bfseries] at (2.15,1.85) {900 baht};
        \node at (2.15,1.25) {per month};
        \node[fill=blue!20,rounded corners,inner sep=7pt] at (2.15,0.55) {Subscribe};
    \end{tikzpicture}
    \end{column}
    \begin{column}{0.52\textwidth}
        \textbf{Ask:} Is 900 baht good value, or does it just look cheap compared with 2,000?
        \medskip
        \begin{itemize}
            \item The original price becomes an \textbf{anchor}.
            \item We may give it too much weight when judging value.
        \end{itemize}
    \end{column}
    \end{columns}
    \begin{block}{Design implication}
        Help users compare features, total costs, and alternatives.
    \end{block}
    {\tiny Illustrative interface. Tversky and Kahneman (1974).}
\end{frame}

\begin{frame}
    \frametitle{2. Confirmation bias}
    \small
    We tend to favor evidence that supports what we already believe.
    \medskip
    \begin{columns}[c]
    \begin{column}{0.44\textwidth}
    \centering
    \begin{tikzpicture}[x=1cm,y=1cm]

        \node[align=center,text width=4cm] at (2.1,3.5)
        {\textbf{``Brand A makes the best phones.''}};
        \draw[rounded corners,fill=blue!5,draw=blue!40] (0.25,0.25) rectangle (3.95,2.8);
        \node at (2.1,2.4) {\textbf{Phone reviews}};
        \node[fill=green!15,rounded corners,inner sep=7pt] at (2.1,1.65) {``Great camera!''};
        \node[text=gray] at (2.1,0.85) {``Battery runs out quickly.''};

    \end{tikzpicture}
    \end{column}
    \begin{column}{0.52\textwidth}

        You already love Brand A.
        \medskip

        You trust the positive review, but dismiss the negative one:
        \begin{quote}
        ``They probably used it wrong.''
        \end{quote}
        \textbf{Ask:} Would you judge the same reviews differently for another brand?

    \end{column}
    \end{columns}
    \begin{block}{Design implication}
Help users compare the same criteria across products. In user testing, actively seek evidence against your own design.
    \end{block}
    {\tiny Illustrative scenario. Nickerson (1998).}
\end{frame}

\begin{frame}
    \frametitle{3. Availability bias}
    \small
    Events that are easy to remember can seem more common than they are.
    \medskip
    \begin{columns}[c]
    \begin{column}{0.44\textwidth}
    \centering
    \begin{tikzpicture}[x=1cm,y=1cm]

        \draw[rounded corners,fill=orange!10,draw=orange!70!black] (0.2,1.5) rectangle (4,3.4);
        \node at (2.1,3) {\textbf{BREAKING NEWS}};
        \node[align=center,text width=3.3cm] at (2.1,2.25) {Dramatic footage\\of a plane accident};
        \draw[->,thick] (2.1,1.4)--(2.1,0.85);
        \node[align=center,text width=4cm] at (2.1,0.35) {``Flying feels dangerous.''};

    \end{tikzpicture}
    \end{column}
    \begin{column}{0.52\textwidth}

        You see repeated footage of a plane accident.
        \medskip

        Before your next flight, accidents suddenly feel very likely---without checking how often they occur.
        \medskip

        \textbf{Ask:} Is the risk high, or is the image easy to recall?

    \end{column}
    \end{columns}
    \begin{block}{Design implication}
When showing risks, provide frequencies and context alongside vivid stories or alerts.
    \end{block}
    {\tiny Illustrative scenario. Tversky and Kahneman (1974).}
\end{frame}

\begin{frame}
    \frametitle{4. Loss aversion}
    \small
    A loss can feel more significant than an equivalent gain, depending on the context.
    \medskip
    \begin{columns}[c]
    \begin{column}{0.44\textwidth}
    \centering
    \begin{tikzpicture}[x=1cm,y=1cm]

        \node[draw=green!50!black,fill=green!10,rounded corners,text width=3.3cm,
        align=center,inner sep=9pt] at (2.1,2.85)
        {Find money\\\Large\textbf{+500 baht}};
        \node[draw=orange!80!black,fill=orange!15,rounded corners,text width=3.3cm,
        align=center,inner sep=9pt] at (2.1,1.1)
        {Lose money\\\Large\textbf{--500 baht}};

    \end{tikzpicture}
    \end{column}
    \begin{column}{0.52\textwidth}

        Imagine finding 500 baht in an old jacket.
        \medskip

        Now imagine losing 500 baht from your wallet.
        \medskip

        \textbf{Ask:} Which feeling would be stronger?
        \medskip

        The loss may hurt more than the equal gain feels good.

    \end{column}
    \end{columns}
    \begin{block}{Design implication}
Make changes reversible: offer undo, backups, and previews so users can explore without risking their work.
    \end{block}
    {\tiny Illustrative scenario. Kahneman, Knetsch, and Thaler (1991).}
\end{frame}

\begin{frame}
    \frametitle{5. Sunk-cost fallacy}
    \small
    Past spending can push us to continue something that is no longer worthwhile.
    \medskip
    \begin{columns}[c]
    \begin{column}{0.44\textwidth}
    \centering
    \begin{tikzpicture}[x=1cm,y=1cm]
\node[align=center,font=\small\bfseries] at (2.1,3.6) {300 baht already paid};
\node[draw,rounded corners,fill=blue!7,text width=3.4cm,align=center,inner sep=8pt] at (2.1,2.55)
{\textbf{Stay}\\Spend another hour\\on a boring movie};
\node[draw,rounded corners,fill=green!10,text width=3.4cm,align=center,inner sep=8pt] at (2.1,0.95)
{\textbf{Leave}\\Use that hour\\for something you enjoy};
\end{tikzpicture}
    \end{column}
    \begin{column}{0.52\textwidth}
You stay because ``I already paid for it.''
\medskip

The ticket money is gone whether you stay or leave.
\medskip

\textbf{Ask:} Is watching the rest worth another hour of your time?
    \end{column}
    \end{columns}
    \begin{block}{Design implication}
When reviewing a design, judge future benefits and costs. Months already spent building it are not a reason to keep a poor solution.
    \end{block}
    {\tiny Arkes and Blumer (1985).}
\end{frame}

\begin{frame}
    \frametitle{6. Dunning--Kruger effect}
    \small
    Limited skill in a task can make it difficult to recognize your own mistakes.
    \medskip
    \begin{center}
\begin{tikzpicture}[x=1cm,y=0.75cm]
\draw[->] (0,0)--(9.8,0) node[right,font=\scriptsize] {Learning};
\draw[->] (0,0)--(0,3.65);
\node[rotate=90,font=\scriptsize] at (-0.45,1.8) {Confidence};
\draw[very thick,blue!70!black] (0.2,0.45) .. controls (0.6,0.6) and (0.65,3.1) .. (1.35,3.1)
 .. controls (2.1,3.1) and (2.3,0.65) .. (3.6,0.65)
 .. controls (5.3,0.65) and (6.1,2.4) .. (9,2.55);
\node[align=center,font=\scriptsize] at (1.5,3.8) {``I know this!''};
\node[align=center,font=\scriptsize] at (3.8,1.6) {``There is more\\than I thought.''};
\node[align=center,font=\scriptsize] at (7.7,3.35) {``I know a lot,\\and I know my limits.''};
\node[below,font=\scriptsize] at (1,0) {Beginner};
\node[below,font=\scriptsize] at (8.5,0) {More experience};
\end{tikzpicture}
\end{center}
{\scriptsize Popular learning-journey sketch, not measured data or the original Dunning--Kruger graph.}
\medskip

\textbf{Key point:} Confidence is not proof of skill. Expertise can bring both confidence and awareness of limits.

    \begin{block}{Design implication}
Provide practice, clear criteria, and feedback. Test actual task performance instead of relying only on self-rated confidence.
    \end{block}
    {\tiny Kruger and Dunning (1999): self-assessment study, not a universal learning trajectory.}
\end{frame}

\begin{frame}
    \frametitle{7. Survivorship bias}
    \small
    We can reach misleading conclusions when we only see the successes.
    \medskip
    \begin{columns}[c]
    \begin{column}{0.44\textwidth}
    \centering
    \begin{tikzpicture}[x=1cm,y=1cm]
\fill[yellow!20] (2.1,3.7)--(0.6,0.7)--(3.6,0.7)--cycle;
\foreach \x in {0.5,1.3,2.1,2.9,3.7} {
 \foreach \y in {0.6,1.6} {
  \fill[gray!40] (\x,\y+0.3) circle (0.12);
  \draw[gray!50,very thick] (\x,\y+0.12)--(\x,\y-0.2);
 }
}
\fill[blue!70!black] (2.1,2.8) circle (0.17);
\draw[blue!70!black,line width=3pt] (2.1,2.6)--(2.1,2.05);
\node[font=\scriptsize,align=center] at (2.1,3.55) {One famous success};
\node[font=\scriptsize,align=center] at (2.1,-0.1) {Other outcomes stay out of view};
\end{tikzpicture}
    \end{column}
    \begin{column}{0.52\textwidth}
``Successful founders dropped out, so dropping out must help you succeed.''
\medskip

This ignores people who made the same choice and did not succeed.
\medskip

\textbf{Ask:} Whose experience is missing?
    \end{column}
    \end{columns}
    \begin{block}{Design implication}
Include people who abandoned your app or failed to finish a task. Interviewing only active users misses important problems.
    \end{block}
    {\tiny Conceptual illustration, not counts. Success stories alone do not establish cause.}
\end{frame}

\begin{frame}
    \frametitle{8. Affect heuristic}
    \small
    We use feelings as a shortcut when judging benefits and risks.
    \medskip
    \begin{columns}[c]
    \begin{column}{0.44\textwidth}
    \centering
    \begin{tikzpicture}[x=1cm,y=1cm]
\draw[rounded corners,thick,fill=black!5] (0.6,0) rectangle (3.6,4.1);
\fill[cyan!15] (0.8,1.5) rectangle (3.4,3.65);
\fill[cyan!45] (0.8,1.5) rectangle (3.4,2.65);
\fill[yellow!30] (0.8,1.5)--(3.4,1.5)--(3.4,2.05) .. controls (2.5,2.5) and (1.8,1.8) .. (0.8,2.15)--cycle;
\fill[yellow!80!orange] (2.85,3.2) circle (0.25);
\node[font=\small\bfseries] at (2.1,3.85) {Dream escape};
\node[fill=blue!65!black,text=white,rounded corners,inner sep=6pt] at (2.1,1.08) {Book now};
\node[font=\tiny,text=gray] at (2.1,0.5) {Seller? Fees? Cancellation?};
\end{tikzpicture}
    \end{column}
    \begin{column}{0.52\textwidth}
You feel excited about a holiday and like the booking app.
\medskip

You assume the booking is safe before checking the seller or cancellation terms.
\medskip

\textbf{Ask:} What evidence supports that feeling?
    \end{column}
    \end{columns}
    \begin{block}{Design implication}
Make fees, cancellation terms, and seller information easy to check. Attractive presentation should support informed decisions.
    \end{block}
    {\tiny Slovic et al. (2007).}
\end{frame}

\begin{frame}
    \frametitle{Activities: Investigate one SIS task}
    \small
    Visit \url{http://sis.ait.ac.th}. Choose one task, such as
    finding a course or checking a registration deadline.
    \begin{block}{1. Walk through the task}
    Note where users might struggle. Use these questions:
    \begin{itemize}
        \item \textbf{Perception and memory:} What is hard to notice,
        read, understand, or remember?
        \item \textbf{Personality and motivation:} Are useful choices
        available? Are progress and next steps clear?
        \item \textbf{Cognition:} What is hard to learn? Where could
        users miss information or make a mistaken decision?
    \end{itemize}
    \end{block}
    \begin{block}{2. Choose five improvements}
    Use \textbf{at least three lecture topics} across your five fixes.
    Choose concepts that explain the problems you find; you do not
    need to use every topic or add gamification.
    \end{block}
    {\footnotesize Describe likely difficulties as predictions unless you have observed users experiencing them.}
\end{frame}

\begin{frame}
    \frametitle{Activities: Explain and test your fixes}
    \small
    Submit \textbf{five pages, one improvement per page}.
    State your chosen task at the top of the first page.
    \begin{enumerate}
        \item \textbf{Problem:} Annotate a screenshot. Explain
        what might go wrong during the task.
        \item \textbf{Concept:} Name a relevant lecture concept
        and explain how it accounts for the difficulty.
        \item \textbf{Fix:} Sketch your change and explain why it helps.
        \item \textbf{Check:} Propose one task and one outcome
        to compare before and after. No experiment is required.
    \end{enumerate}
    \begin{block}{Example: A deadline is easy to miss}
    \textbf{Attention:} The deadline blends into unrelated dates.
    \textbf{Fix:} Put it beside the registration action with a clear label.
    \textbf{Check:} Ask users to find the deadline; compare correct answers.
    \end{block}
    {\footnotesize Aim for clear reasoning and useful changes, not the number of theory names.}
\end{frame}
